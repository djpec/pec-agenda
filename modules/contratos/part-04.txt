      if(diff <= 0){
        return { valid:false, text:'Verifique as datas e horários', minutes:diff };
      }

      const hours = Math.floor(diff / 60);
      const minutes = diff % 60;

      if(hours) text += `${hours} ${hours === 1 ? 'hora' : 'horas'}`;
      if(minutes) text += `${hours ? ' e ' : ''}${minutes} ${minutes === 1 ? 'minuto' : 'minutos'}`;
