function renderBindings(){
  document.querySelectorAll("[data-bind]").forEach(el => {

    if(el.dataset.bind === "issueDate" || el.dataset.bind === "eventDate") v = dateBR(v);
    el.textContent = v;
  });

    const v = data[k] || "—";
    let href = "";

    if(k === "email" && v !== "—") href = `mailto:${v}`;
    if(k === "site" && v !== "—") href = /^https?:\/\//i.test(v) ? v : `https://${v}`;
    if(k === "instagram" && v !== "—"){
      const u = v.replace(/^@/,"").replace(/\/$/,"");
      href = `https://www.instagram.com/${u}/`;
    }

    el.innerHTML = "";
    if(!href){
      el.textContent = v;
      return;
    }

    const a = document.createElement("a");

    a.textContent = v;
    a.href = href;

      a.rel = "noopener noreferrer";
    }
    el.appendChild(a);
  });
}
