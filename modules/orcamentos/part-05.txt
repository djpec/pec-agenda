function highlightCurrency(text){
  const safe = esc(text || "—");
  return safe.replace(/(R\$\s?[\d\.\,]+)/g, "<strong>$1</strong>");
}

function parseMoneyBR(value){
  const text = String(value || "").trim();
  const match = text.match(/R\$\s*([\d\.\,]+)/i) || text.match(/([\d\.\,]+)/);
  if(!match) return null;

  let num = match[1];

  if(num.includes(",")){
    num = num.replace(/\./g, "").replace(",", ".");
  }else{
    const parts = num.split(".");
    if(parts.length > 2){
      num = parts.join("");
    }else if(parts.length === 2 && parts[1].length === 3){
      num = parts.join("");
    }
  }

  const parsed = Number(num);
  return Number.isFinite(parsed) ? parsed : null;
}

function formatMoneyBR(value){
  if(!Number.isFinite(value)) return "";
  return value.toLocaleString("pt-BR", {
    style:"currency",
    currency:"BRL",
    minimumFractionDigits:2,
    maximumFractionDigits:2
  });
}

function getBaseInvestmentValue(){
  if(data.investmentMode === "perItem"){
    const values = data.scope
      .map(item => parseMoneyBR(item.value))
      .filter(v => Number.isFinite(v));

    if(!values.length) return null;
    return values.reduce((sum, value) => sum + value, 0);
  }

  return parseMoneyBR(data.total);
}

function formatOptionalValue(value){
  const text = String(value || "").trim();
  if(!text) return "—";

  const exactMoney = text.match(/^\s*R\$\s*[\d\.\,]+\s*$/i);

  if(exactMoney){
    return `+ ${esc(text)}`;
  }

  return highlightCurrency(text);
}

const PAGE_FORMATS = {
  a5:     {label:"A5", width:148,   height:210},
  a4:     {label:"A4", width:210,   height:297},
  letter: {label:"Carta", width:215.9, height:279.4},

};

function getPageDimensions(){
  const base = PAGE_FORMATS[data.pageSize] || PAGE_FORMATS.a4;
  const landscape = data.orientation === "landscape";

  return landscape
    ? {width:base.height, height:base.width, label:base.label}
    : {width:base.width, height:base.height, label:base.label};
}

function ensureDynamicPageStyle(){
  let style = document.getElementById("dynamicPageStyle");

  if(!style){
    style = document.createElement("style");
    style.id = "dynamicPageStyle";
    document.head.appendChild(style);
  }

  const page = getPageDimensions();

  style.textContent = `
    @page {
      size: ${page.width}mm ${page.height}mm;
      margin: 10mm 11mm 19mm;
    }
  `;
}

function renderPageFormat(){
  const page = getPageDimensions();

  document.documentElement.style.setProperty("--sheet-width", `${page.width}mm`);
  document.documentElement.style.setProperty("--sheet-min-height", `${page.height}mm`);

  /*
    Só tratamos como estreita quando a largura útil é realmente pequena.
    Carta/A4 continuam com o layout normal.
  */
  document.body.dataset.paperNarrow = page.width < 185 ? "true" : "false";

  ensureDynamicPageStyle();
}

function pxPerMmForPrint(){
  const probe = document.createElement("div");
  probe.style.position = "absolute";
  probe.style.visibility = "hidden";
  probe.style.width = "100mm";
  probe.style.height = "1px";
  document.body.appendChild(probe);
  const px = probe.getBoundingClientRect().width / 100;
  probe.remove();
  return px;
}

function resetFinalFooterGap(){
  const footer = document.getElementById("finalFooter");
  if(footer){
    footer.style.removeProperty("--print-footer-gap");
  }
}

function pinFooterToLastPrintedPage(){
  resetFinalFooterGap();

  const doc = document.getElementById("doc");
  const footer = document.getElementById("finalFooter");

  if(!doc || !footer) return;

  const page = getPageDimensions();
  const mm = pxPerMmForPrint();

  /* Margens do @page: 10mm superior + 13mm inferior. */
  const printableHeight = (page.height - 23) * mm;

  if(printableHeight <= 0) return;

  const docRect = doc.getBoundingClientRect();
  const footerRect = footer.getBoundingClientRect();

  const docTop = docRect.top + window.scrollY;
  const footerTop = (footerRect.top + window.scrollY) - docTop;
  const footerHeight = footerRect.height;

  /*
    Descobre em qual folha o rodapé começa e quanto espaço resta.
    Se couber nessa folha, usamos só o espaço necessário para encostar
    o rodapé no final. Se não couber, ele vai para a próxima e é
    posicionado na base dela.
  */
  const remainder = ((footerTop % printableHeight) + printableHeight) % printableHeight;
  const breathing = 4 * mm;

  let gap;

  if(remainder + footerHeight + breathing <= printableHeight){
    gap = printableHeight - remainder - footerHeight;
  }else{
    gap = (printableHeight - remainder) + printableHeight - footerHeight;
  }

  gap = Math.max(16, gap);
  footer.style.setProperty("--print-footer-gap", `${gap}px`);
}
