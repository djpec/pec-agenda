function autoDensity(){
  let score = 0;
  score += data.scope.length * 2;
  score += data.extras.length * 2;
  score += String(data.notes || "").length > 180 ? 2 : 0;
  score += String(data.rider || "").split("\n").filter(Boolean).length > 4 ? 2 : 0;
  score += data.scope.reduce((a,s) => a + String(s.description || "").length, 0) > 350 ? 2 : 0;

  const doc = document.getElementById("doc");
  doc.dataset.density = score >= 10 ? "compact" : "normal";
}

function xmlEsc(v){
  return String(v ?? "")
    .replaceAll("&","&amp;")
    .replaceAll("<","&lt;")
    .replaceAll(">","&gt;")
    .replaceAll('"',"&quot;")
    .replaceAll("'","&apos;");
}

  const keys = ["theme","palette","pageSize","orientation","pdfScale","investmentMode","client","issueDate","eventDate","venue","city","time","duration","musicStyle","intro","total","totalDesc","deposit","balance","payment","notes","rider","phone","email","site","instagram"];

      <duration>${xmlEsc(s.duration)}</duration>
      <time>${xmlEsc(s.time)}</time>
      <value>${xmlEsc(s.value || "")}</value>
    </item>`).join("");
  const extras = data.extras.map(e => `
    <extra>
      <title>${xmlEsc(e.title)}</title>
      <description>${xmlEsc(e.description)}</description>
      <value>${xmlEsc(e.value)}</value>
      <ratio>${xmlEsc(e.ratio || "4/3")}</ratio>
      <fit>${xmlEsc(e.fit || "cover")}</fit>
      <position>${xmlEsc(e.position || "center")}</position>
    </extra>`).join("");

  const xml = `<?xml version="1.0" encoding="UTF-8"?>
<proposal>
${fixed}
  <showExtras>${data.showExtras}</showExtras>
  <scope>${scope}
  </scope>
  <extras>${extras}
  </extras>
</proposal>`;

  const blob = new Blob([xml], {type:"application/xml;charset=utf-8"});
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = `proposta-${String(data.client || "cliente").toLowerCase().replace(/\s+/g,"-")}.xml`;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

function importXML(file){
  const r = new FileReader();
  r.onload = () => {
    try{
      const xml = new DOMParser().parseFromString(r.result,"application/xml");
      if(xml.querySelector("parsererror")) throw new Error();
      const root = xml.documentElement;
      const n = clone(DEFAULT);
      const keys = ["theme","palette","pageSize","orientation","pdfScale","investmentMode","client","issueDate","eventDate","venue","city","time","duration","musicStyle","intro","total","totalDesc","deposit","balance","payment","notes","rider","phone","email","site","instagram"];
      keys.forEach(k => {
        const el = root.querySelector(`:scope > ${k}`);
        if(el) n[k] = el.textContent;
      });
      const show = root.querySelector(":scope > showExtras");
      if(show) n.showExtras = show.textContent === "true";

      const items = [...root.querySelectorAll(":scope > scope > item")];
      if(items.length){
        n.scope = items.map(s => ({
          title:text(s,"title"),
          description:text(s,"description"),
          duration:text(s,"duration"),
          time:text(s,"time"),
          value:text(s,"value")
        }));
      }

      const ex = [...root.querySelectorAll(":scope > extras > extra")];
      if(ex.length){
        n.extras = ex.map(e => normalizeExtra({
          title:text(e,"title"),
          description:text(e,"description"),
          value:text(e,"value"),
          ratio:text(e,"ratio") || "4/3",
          fit:text(e,"fit") || "cover",
          position:text(e,"position") || "center",
          image:""
        }));
      }

      if(!n.palette) n.palette = "terracota";

      data = n;
      fillFixed();
      renderScopeEditor();
      renderExtrasEditor();
      renderAll();
      alert("Dados importados.");
    }catch{
      alert("Não foi possível importar o XML.");
    }
  };
  r.readAsText(file,"UTF-8");
}
